cmake_minimum_required(VERSION 3.24)
project(NocturneWindowsHost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(nocturne_host SHARED capture.cpp devices.cpp decode.cpp desktop_audio.cpp)
target_link_libraries(nocturne_host PRIVATE d3d11 dxgi mfplat mf mfuuid wmcodecdspuuid ole32 propsys windowsapp gdi32 mmdevapi)
target_compile_options(nocturne_host PRIVATE /W4 /EHsc)
include(CTest)
add_executable(nocturne-native-tests tests.cpp capture.cpp devices.cpp decode.cpp desktop_audio.cpp)
target_link_libraries(nocturne-native-tests PRIVATE d3d11 dxgi mfplat mf mfuuid wmcodecdspuuid ole32 propsys windowsapp gdi32 mmdevapi dwmapi wtsapi32 shell32)
target_compile_options(nocturne-native-tests PRIVATE /W4 /EHsc)
add_test(NAME NativeWindowsMedia COMMAND nocturne-native-tests)
add_test(NAME NativeWindowCapture COMMAND nocturne-native-tests --window-capture)
add_test(NAME NativeTrayRegistration COMMAND nocturne-native-tests --tray)
set_tests_properties(NativeWindowCapture PROPERTIES SKIP_RETURN_CODE 77)
set_tests_properties(NativeTrayRegistration PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 15)
